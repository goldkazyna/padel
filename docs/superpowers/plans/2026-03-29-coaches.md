# Тренеры клуба — План реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Добавить управление тренерами клуба с расписанием и привязкой к бронированиям кортов.

**Architecture:** Тренер — пользователь с ролью `coach`, привязанный к клубу через pivot-таблицу `club_coaches`. У тренера есть недельный шаблон расписания (`coach_schedules`) и исключения на конкретные даты (`coach_schedule_overrides`). При бронировании корта админ может выбрать свободного тренера. Бронь автоматически блокирует тренера на это время.

**Tech Stack:** Laravel 12, Blade, Alpine.js, Bootstrap 5, SQLite

```php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class ClubCoach extends Model
{
    protected $fillable = [
        'club_id',
        'user_id',
        'specialization',
        'hourly_rate',
    ];

    protected $casts = [
        'hourly_rate' => 'decimal:2',
    ];

    public function club()
    {
        return $this->belongsTo(Club::class);
    }

    public function user()
    {
        return $this->belongsTo(User::class);
    }

    public function schedules()
    {
        return $this->hasMany(CoachSchedule::class);
    }

    public function overrides()
    {
        return $this->hasMany(CoachScheduleOverride::class);
    }

    /**
     * Проверяет, работает ли тренер в указанную дату и время.
     */
    public function isAvailableAt(string $date, string $startTime, string $endTime): bool
    {
        $dayOfWeek = \Carbon\Carbon::parse($date)->dayOfWeekIso; // 1=пн, 7=вс

        // Проверяем исключения на эту дату
        $override = $this->overrides()->where('date', $date)->first();
        if ($override) {
            if (!$override->is_available) {
                // Выходной на эту дату
                if (!$override->start_time) return false; // весь день выходной
                // Частичный выходной — проверяем пересечение
                if ($startTime < $override->end_time && $endTime > $override->start_time) {
                    return false;
                }
            } else {
                // Дополнительный рабочий день
                if ($override->start_time && $override->end_time) {
                    return $startTime >= $override->start_time && $endTime <= $override->end_time;
                }
                return true;
            }
        }

        // Проверяем недельный шаблон
        $schedule = $this->schedules()->where('day_of_week', $dayOfWeek)->first();
        if (!$schedule) return false;

        return $startTime >= $schedule->start_time && $endTime <= $schedule->end_time;
    }

    /**
     * Проверяет, свободен ли тренер (нет пересечения с другими бронями).
     */
    public function isFreeAt(string $date, string $startTime, string $endTime): bool
    {
        if (!$this->isAvailableAt($date, $startTime, $endTime)) {
            return false;
        }

        // Проверяем нет ли бронирований на это время
        $hasBooking = CourtBooking::where('coach_id', $this->user_id)
            ->whereDate('date', $date)
            ->where('status', 'confirmed')
            ->where(function ($q) use ($startTime, $endTime) {
                $q->where('start_time', '<', $endTime)
                  ->where('end_time', '>', $startTime);
            })
            ->exists();

        return !$hasBooking;
    }
}
```

### Task 3: Контроллер тренеров — CRUD + расписание

**Files:**

- [ ] **Step 1: Создать контроллер**

```php
<?php

namespace App\Http\Controllers\Club;

use App\Http\Controllers\Controller;
use App\Models\ClubCoach;
use App\Models\CoachSchedule;
use App\Models\CoachScheduleOverride;
use App\Models\User;
use Illuminate\Http\Request;

class CoachController extends Controller
{
    private function getClub()
    {
        $user = auth()->user();
        if ($user->isSuperAdmin()) return \App\Models\Club::first();
        if ($user->isClubModerator()) return $user->moderatorClubs()->first();
        return $user->adminClubs()->first();
    }

    public function index()
    {
        $club = $this->getClub();
        if (!$club) return redirect()->route('club.dashboard')->with('error', 'Клуб не найден');

        $clubCoaches = ClubCoach::where('club_id', $club->id)
            ->with(['user', 'schedules'])
            ->get();

        return view('club.coaches.index', compact('clubCoaches', 'club'));
    }

    public function store(Request $request)
    {
        $club = $this->getClub();
        if (!$club) return back()->with('error', 'Клуб не найден');

        $validated = $request->validate([
            'user_id' => 'required|exists:users,id',
            'specialization' => 'nullable|string|max:255',
            'hourly_rate' => 'nullable|numeric|min:0',
        ]);

        $user = User::findOrFail($validated['user_id']);
        $user->update(['role' => 'coach']);

        $clubCoach = ClubCoach::create([
            'club_id' => $club->id,
            'user_id' => $validated['user_id'],
            'specialization' => $validated['specialization'] ?? null,
            'hourly_rate' => $validated['hourly_rate'] ?? null,
        ]);

        return back()->with('success', 'Тренер добавлен');
    }

    public function update(Request $request, User $user)
    {
        $club = $this->getClub();
        $clubCoach = ClubCoach::where('club_id', $club->id)->where('user_id', $user->id)->firstOrFail();

        $validated = $request->validate([
            'specialization' => 'nullable|string|max:255',
            'hourly_rate' => 'nullable|numeric|min:0',
        ]);

        $clubCoach->update($validated);

        return back()->with('success', 'Тренер обновлён');
    }

    public function destroy(User $user)
    {
        $club = $this->getClub();
        $clubCoach = ClubCoach::where('club_id', $club->id)->where('user_id', $user->id)->firstOrFail();
        $clubCoach->delete();

        if ($user->coachProfile()->count() === 0) {
            $user->update(['role' => 'player']);
        }

        return back()->with('success', 'Тренер удалён');
    }

    // --- Расписание ---

    public function schedule(User $user)
    {
        $club = $this->getClub();
        $clubCoach = ClubCoach::where('club_id', $club->id)
            ->where('user_id', $user->id)
            ->with(['schedules', 'overrides', 'user'])
            ->firstOrFail();

        $dayNames = ['', 'Понедельник', 'Вторник', 'Среда', 'Четверг', 'Пятница', 'Суббота', 'Воскресенье'];

        return view('club.coaches.schedule', compact('clubCoach', 'dayNames'));
    }

    public function updateSchedule(Request $request, User $user)
    {
        $club = $this->getClub();
        $clubCoach = ClubCoach::where('club_id', $club->id)->where('user_id', $user->id)->firstOrFail();

        $validated = $request->validate([
            'schedules' => 'nullable|array',
            'schedules.*.day_of_week' => 'required|integer|min:1|max:7',
            'schedules.*.start_time' => 'required|date_format:H:i',
            'schedules.*.end_time' => 'required|date_format:H:i',
        ]);

        $clubCoach->schedules()->delete();

        foreach ($validated['schedules'] ?? [] as $s) {
            CoachSchedule::create([
                'club_coach_id' => $clubCoach->id,
                'day_of_week' => $s['day_of_week'],
                'start_time' => $s['start_time'],
                'end_time' => $s['end_time'],
            ]);
        }

        return back()->with('success', 'Расписание обновлено');
    }

    public function addOverride(Request $request, User $user)
    {
        $club = $this->getClub();
        $clubCoach = ClubCoach::where('club_id', $club->id)->where('user_id', $user->id)->firstOrFail();

        $validated = $request->validate([
            'date' => 'required|date',
            'start_time' => 'nullable|date_format:H:i',
            'end_time' => 'nullable|date_format:H:i',
            'is_available' => 'required|boolean',
            'reason' => 'nullable|string|max:255',
        ]);

        // Удаляем старый override на эту дату если есть
        $clubCoach->overrides()->where('date', $validated['date'])->delete();

        CoachScheduleOverride::create([
            'club_coach_id' => $clubCoach->id,
            'date' => $validated['date'],
            'start_time' => $validated['start_time'] ?? null,
            'end_time' => $validated['end_time'] ?? null,
            'is_available' => $validated['is_available'],
            'reason' => $validated['reason'] ?? null,
        ]);

        return back()->with('success', $validated['is_available'] ? 'Дополнительный день добавлен' : 'Выходной добавлен');
    }

    public function deleteOverride(CoachScheduleOverride $override)
    {
        $override->delete();
        return back()->with('success', 'Исключение удалено');
    }

    // Поиск пользователей для добавления тренера
    public function searchUsers(Request $request)
    {
        $query = $request->get('q', '');
        if (strlen($query) < 2) return response()->json([]);

        $club = $this->getClub();
        $existingCoachIds = ClubCoach::where('club_id', $club->id)->pluck('user_id');

        $users = User::where(function ($q) use ($query) {
                $q->where('first_name', 'like', "%{$query}%")
                  ->orWhere('last_name', 'like', "%{$query}%")
                  ->orWhere('phone', 'like', "%{$query}%")
                  ->orWhere('email', 'like', "%{$query}%");
            })
            ->whereNotIn('id', $existingCoachIds)
            ->limit(10)
            ->get()
            ->map(fn($u) => [
                'id' => $u->id,
                'name' => $u->full_name,
                'phone' => $u->phone,
                'email' => $u->email,
            ]);

        return response()->json($users);
    }
}
```
